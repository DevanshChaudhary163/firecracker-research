import csv
import statistics
from pathlib import Path

folder = Path("results/benchmarks")

for method in ["snapshot", "uffd", "reap"]:
    path = folder / f"{method}.csv"

    with path.open() as f:
        rows = list(csv.DictReader(f))

    times = [float(r["total_ms"]) for r in rows]
    ordered = sorted(times)

    # Nearest-rank P95, matching the earlier analysis script
    p95_index = max(0, (95 * len(times) + 99) // 100 - 1)

    print(f"\n{method.upper()}")
    print(f"Trials: {len(times)}")
    print(f"Mean: {statistics.mean(times):.3f} ms")
    print(f"Median: {statistics.median(times):.3f} ms")
    print(f"Minimum: {min(times):.3f} ms")
    print(f"Maximum: {max(times):.3f} ms")
    print(f"P95: {ordered[p95_index]:.3f} ms")
    if len(times) > 1:
        print(f"Std dev: {statistics.stdev(times):.3f} ms")
